V16: no attitude CDC constraints
board/constraints/hz_zynq_imu_clock_groups.xdc
tt1# BEGIN V17 ATTITUDE CDC (append-only; retain existing FIFO clock groups)
2# MPU FCLK1 (50 MHz) and FCLK0 (100 MHz) have no shared STA clock node.
3# Sequence: double synchronization, then 3 stable clocks; the six-axis bus
4# is held until the next 10 ms sample. Physical data transfer must be below
5# 20 ns (settling window >40 ns); bus skew must be below 10 ns.
6# These are bounded held-bus CDC paths, not synchronous FCLK paths.
7set imu_meta [get_cells -hier -filter {NAME =~ mpu_attitude_0/inst/sequence_meta_reg*}]
8set imu_bus [get_cells -hier -filter {NAME =~ mpu_attitude_0/inst/sensor_?_reg*}]
9set imu_all [get_cells -hier -filter {NAME =~ mpu_attitude_0/inst/sequence_meta_reg* || NAME =~ mpu_attitude_0/inst/sensor_?_reg*}]
10set imu_starts [get_cells -hier -filter {NAME =~ mpu6050_tilt_0/*/mpu_data_reg* || NAME =~ mpu6050_tilt_0/*/motion_?_reg*}]
11set imu_data_starts [get_cells -hier -filter {NAME =~ mpu6050_tilt_0/*/motion_?_reg*}]
12set_max_delay -datapath_only 20.0 -from $imu_starts -to $imu_all
13set_bus_skew 10.0 -from $imu_data_starts -to $imu_bus
14# END V17 ATTITUDE CDC
